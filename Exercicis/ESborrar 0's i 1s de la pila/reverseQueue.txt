#include <iostream>
#include <queue>

using namespace std;

void reverseQueue_aux(queue<int> &q)
{
	if(not(q.empty()))
    {
		int k = q.front();
		q.pop();
		reverseQueue_aux(q);
		q.push(k);
	}
}
queue<int> reverseQueue(queue<int> q)
{
	reverseQueue_aux(q);
	return q;
}

int main()
{
	queue<int> q;
	int a;
	while (cin >> a)
	{
		q.push(a);
	}
	q = reverseQueue(q);
	while (not(q.empty()))
	{
		int t = q.front();
		q.pop();
		cout << t << " ";
	}
}
